def play_game(model, qmaze, pirate_cell, max_steps=None):
    qmaze.reset(pirate_cell)
    envstate = qmaze.observe()
    steps = 0
    if max_steps is None:
        max_steps = qmaze.maze.size * 4  # safety cutoff

    while steps < max_steps:
        state = np.asarray(envstate, dtype=np.float32)
        if state.ndim == 1:
            state = np.expand_dims(state, axis=0)

        q_values = model(state, training=False).numpy()
        action = np.argmax(q_values[0])

        envstate, reward, game_status = qmaze.act(action)
        steps += 1

        if game_status == 'win':
            return True
        elif game_status == 'lose':
            return False

    return False  # timed out with no result

def completion_check(model, maze_or_qmaze, max_steps=None):
    # Accept either raw numpy maze or TreasureMaze instance
    if isinstance(maze_or_qmaze, TreasureMaze):
        qmaze = maze_or_qmaze
    else:
        qmaze = TreasureMaze(maze_or_qmaze)

    for cell in qmaze.free_cells:
        if not qmaze.valid_actions(cell):
            continue
        if not play_game(model, qmaze, cell, max_steps=max_steps):
            return False
    return True

def build_model(maze):
    model = Sequential()
    model.add(Dense(maze.size, input_shape=(maze.size,)))

    model.add(Dense(maze.size))
    model.add(PReLU())
    model.add(Dense(num_actions))
    model.compile(optimizer='adam', loss='mse')
    return model

def train_step(x, y):
    with tf.GradientTape() as tape:
        q_values = model(x, training=True)
        loss = loss_fn(y, q_values)
    grads = tape.gradient(loss, model.trainable_variables)
    optimizer.apply_gradients(zip(grads, model.trainable_variables))
    return loss

def qtrain(model, maze, **opt):
    # exploration factor
    global epsilon 
    
    # Number of epochs
    n_epoch = opt.get('n_epoch', 15000)
    
    # Maximum memory to store episodes
    max_memory = opt.get('max_memory', 1000)
    
    # Maximum data size for training
    data_size = opt.get('data_size', 50)
    
    # Frequency of target network updates
    target_update_freq = opt.get('target_update_freq', 50)
    
    # Start time
    start_time = datetime.datetime.now()
    
    # Construct environment/game from numpy array: maze (see argument above)
    qmaze = TreasureMaze(maze)
    
    # Target Network to better guide training
    target_model = clone_model(model)
    target_model.set_weights(model.get_weights())
    
    # Initialize experience replay object
    experience = GameExperience(model, target_model, max_memory=max_memory)

    win_history = [] # history of win/lose game
    hsize = qmaze.maze.size // 2 # history window size
    win_rate = 0.0
    
    # =============START_HERE================
    for epoch in range(n_epoch):
        #Reset the environment at a random starting cell
        agent_cell = random.choice(qmaze.free_cells)
        qmaze.reset(agent_cell)
        
        #Set the initial environment state
        env_state = qmaze.observe()
        
        loss = 0.0
        n_episodes = 0
        game_over = False

        # While game status is not game over
        while not game_over:
            previous_envstate = env_state
            
            #Decide on an action (Epsilon-Greedy Strategy)
            if np.random.rand() < epsilon:
                # Explore: take a random valid exploration action
                action = random.choice(qmaze.valid_actions())
            else:
                # Exploit: pick the best action prediction from the experience object
                prediction = experience.predict(previous_envstate)
                
                # Dynamic shape handling for array outputs vs single scalar indices
                prediction = np.squeeze(np.array(prediction))
                if prediction.ndim > 0 and len(prediction) > 1:
                    # experience.predict returned raw Q-values list, grab index of max reward
                    action = np.argmax(prediction)
                elif prediction.ndim == 0:
                    # experience.predict returned a single value object scalar
                    action = prediction
                else:
                    # Multi-dimensional batch squeeze array fallback
                    action = np.argmax(prediction)

            # Explicitly cast to a native Python int scalar.
            # This prevents ValueError exceptions within the TreasureMaze framework!
            action = int(action)

            #Retrieve values from the act() method
            env_state, reward, game_status = qmaze.act(action)
            n_episodes += 1
            
            #Track the wins and losses from the game_status using win_history 
            if game_status == 'win':
                win_history.append(1)
                game_over = True
            elif game_status == 'lose':
                win_history.append(0)
                game_over = True

            #Store the episode in the Experience replay object
            # Using exact naming requested in the snippet instructions
            episode = [previous_envstate, action, reward, env_state, game_status]
            experience.remember(episode)
            
            #Train neural network model and evaluate loss
            # Only train if we have accumulated enough exploration steps to form a clean batch
            if len(experience.memory) >= data_size:
                # Call GameExperience.get_data to retrieve training data (input and target)
                inputs, targets = experience.get_data(batch_size=data_size)
                
                # Pass to the train_step method, assign to batch_loss, and add to loss
                # Note: If you opted for the explicit model fix from earlier, change to train_step(model, inputs, targets)
                batch_loss = train_step(inputs, targets)
                
                # Safely extract scalar value from TensorFlow tensor if needed
                if hasattr(batch_loss, 'numpy'):
                    loss += batch_loss.numpy()
                else:
                    loss += batch_loss

        # Periodically update target network variables to stabilize variance 
        if epoch % target_update_freq == 0:
            target_model.set_weights(model.get_weights())

        # Win rate over last hsize steps 
        win_rate = sum(win_history[-hsize:]) / hsize if len(win_history) >= hsize else 0.0

        # Print the epoch, loss, episode, win count, win rate, and time for each epoch
        dt = datetime.datetime.now() - start_time
        t = format_time(dt.total_seconds())
        
        # Calculate running average epoch loss based on total movements taken
        epoch_loss = loss / n_episodes if n_episodes > 0 else 0.0
        
        print("Epoch: {:03d}/{:d} | Loss: {:.4f} | Episodes: {:d} | Win count: {:d} | Win rate: {:.3f} | time: {}".format(
            epoch, n_epoch-1, epoch_loss, n_episodes, sum(win_history), win_rate, t))

        # Epsilon Decay Mechanics
        if win_rate > 0.9:
            epsilon = 0.05
        else:
            epsilon = max(epsilon * epsilon_decay, epsilon_min)
    
        # If the win rate is above threshold and model passes completion check
        if win_rate >= 0.999 and completion_check(model, maze):
            print(f"Reached 100% win rate and passed absolute verification at epoch {epoch}!")
            break

    total_time = format_time((datetime.datetime.now() - start_time).total_seconds())
    print("Training complete in:", total_time)
    return win_rate

def format_time(seconds):
    if seconds < 400:
        s = float(seconds)
        return "%.1f seconds" % (s,)
    elif seconds < 4000:
        m = seconds / 60.0
        return "%.2f minutes" % (m,)
    else:
        h = seconds / 3600.0
        return "%.2f hours" % (h,)
